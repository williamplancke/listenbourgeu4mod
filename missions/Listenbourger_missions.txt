lst_missions_iberian_dominance = {
    slot = 1
    generic = no
    ai = yes
    potential = {
        tag = LST
        NOT = { map_setup = map_setup_random }
    }
    has_country_shield = yes
    lst_rally_the_forces = {
        #icon
        trigger = {
            army_size_percentage = 1
            num_of_generals = 1
        }
        effect = {
            leon_area = {
                limit = {
                    NOT = { is_core = ROOT }
                    NOT = { is_permanent_claim = ROOT }
                }
                add_permanent_claim = ROOT
            }
            beieras_area = {
                limit = {
                    NOT = { is_core = ROOT }
                    NOT = { is_permanent_claim = ROOT }
                }
                add_permanent_claim = ROOT
            }
            define_advisor = {
                type = commandant
                skill = 1
                cost_multiplier = 0.75
            }
        }
    }
    lst_charge_into_iberia = {
        #icon
        required_missions = {lst_rally_the_forces}
        trigger = {
            OR = {
                AND = {
                    leon_area = {
                        type = all
                        owned_by = ROOT
                    }
                    beieras_area = {
                        type = all
                        OR = {
                            owned_by = ROOT
                            owned_by_subject_of = ROOT
                        }
                    }
                }
                total_development = 350
            }
        }
        effect = {
            if = {
                limit = {
                    leon_area = {
                        type = all
                        owned_by = ROOT
                    }
                    leon_area = {
                        type = all
                        owned_by = ROOT
                    }
                }
                add_country_modifier = {
                    name = "lst_meritocratic_administration"
                    duration = -1
                }
            }
            if = {
                limit = {
                    total_development = 350
                }
                add_country_modifier = {
                    name = "lst_meritocratic_development"
                    duration = -1
                }
            }
            custom_tooltip = lst_charge_into_iberia_effect_tt
        }
    }
    lst_break_the_spanish_crown = {
        # icon
        required_missions = {lst_charge_into_iberia}
        trigger = {
            num_of_owned_provinces_with = {
                region = region_iberia
                amount = 40
            }
            OR = {
                227 = {
                    owned_by = ROOT
                    is_core = ROOT
                }
                219 = {
                    owned_by = ROOT
                    is_core = ROOT
                }
                213 = {
                    owned_by = ROOT
                    is_core = ROOT
                }
            }
        }
        effect = {
            country_event = {
                id = flavor_lst.6
            }
        }
    }
}
